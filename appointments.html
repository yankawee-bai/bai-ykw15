-- ============================================================ -- appointments.sql -- ตารางนัดหมายแพทย์ของผู้สูงอายุแต่ละคน -- ใช้สิทธิ์ร่วมกับฟังก์ชัน can_manage_elderly() ที่สร้างไว้แล้วใน -- medications.sql (เจ้าของ หรือผู้ดูแล/บุคลากรที่เชื่อมโยงแบบ -- accepted จัดการได้หมด) -- รันใน Supabase SQL editor (ครั้งเดียว ไม่ต้องรันซ้ำ) -- ต้องรัน caregiver_links.sql + caregiver_links_patch_ui.sql -- และ medications.sql (เพื่อให้มีฟังก์ชัน can_manage_elderly) มาก่อน -- ============================================================ create table if not exists public.appointments ( id uuid primary key default gen_random_uuid(), patient_id uuid not null references public.profiles(id) on delete cascade, created_by uuid not null references public.profiles(id) on delete cascade, title text not null, doctor_name text, location text, appointment_at timestamptz not null, notes text, status text not null default 'scheduled' check (status in ('scheduled', 'completed', 'cancelled')), created_at timestamptz not null default now() ); comment on table public.appointments is 'นัดหมายแพทย์ของผู้สูงอายุแต่ละคน จัดการได้โดยเจ้าตัวหรือผู้ดูแล/บุคลากรที่เชื่อมโยง (accepted) แล้ว'; grant select, insert, update, delete on public.appointments to authenticated; alter table public.appointments enable row level security; -- ------------------------------------------------------------ -- trigger กันข้อมูลผิด role (patient_id ต้องเป็น role = elderly) -- สร้างฟังก์ชันแยกต่างหาก เพราะตารางนี้ใช้ชื่อคอลัมน์ patient_id -- ไม่ใช่ elderly_id แบบตาราง medications/medication_logs -- ------------------------------------------------------------ create or replace function public.enforce_appointment_patient_role() returns trigger language plpgsql security definer set search_path = public as $$ begin if not exists ( select 1 from public.profiles where id = new.patient_id and role = 'elderly' ) then raise exception 'patient_id ต้องเป็นโปรไฟล์ที่มี role = elderly'; end if; return new; end; $$; drop trigger if exists appointments_patient_role_check on public.appointments; create trigger appointments_patient_role_check before insert or update on public.appointments for each row execute function public.enforce_appointment_patient_role(); -- ------------------------------------------------------------ -- RLS policies — ใช้ can_manage_elderly() ตัวเดียวกับ medications -- ------------------------------------------------------------ drop policy if exists "Can view appointments of linked elderly" on public.appointments; create policy "Can view appointments of linked elderly" on public.appointments for select using (public.can_manage_elderly(patient_id)); drop policy if exists "Can add appointments for linked elderly" on public.appointments; create policy "Can add appointments for linked elderly" on public.appointments for insert with check ( public.can_manage_elderly(patient_id) and created_by = auth.uid() ); drop policy if exists "Can update appointments of linked elderly" on public.appointments; create policy "Can update appointments of linked elderly" on public.appointments for update using (public.can_manage_elderly(patient_id)) with check (public.can_manage_elderly(patient_id)); drop policy if exists "Can delete appointments of linked elderly" on public.appointments; create policy "Can delete appointments of linked elderly" on public.appointments for delete using (public.can_manage_elderly(patient_id));
